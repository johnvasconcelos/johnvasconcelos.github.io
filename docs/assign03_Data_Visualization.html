1. Document a Chart

library(ggplot2)

# Creating the data frame with the data: proportion of prefered pie flavors)
sales <- data.frame(
  flavor = c("Blueberry", "Cherry", "Apple", "Boston Cream", "Other", "Vanilla"),
  prop  = c(0.12, 0.3, 0.26, 0.16, 0.04, 0.12)
)

# Keeping the original order of the flavores (to keep ggplot from sorting 
# alphabetically)
sales$sabor <- factor(sales$flavor, levels = sales$flavor)

# Plotting
# Ggplot2 has no pie chart. We need to create a col chart and roll it in
ggplot(sales, aes(x = 1, y = prop, fill = flavor)) +
# Plotting a single stacked bar
  geom_col(width = 1, color = "black",
           position = position_stack(reverse = TRUE)) +
# Placing each flavor's name on the outside and middle of each slice
  geom_text(aes(x = 1.7, label = sabor),
            position = position_stack(vjust = 0.5, reverse = TRUE), size = 3) +
  # Wrapping the bar in a circle, from 3 o'clock and spinning anti-clock
  coord_polar(theta = "y", start = pi / 2, direction = -1) +
  # Using 6 shades of gray from 0.3 to 1 where 1 is totally white
  scale_fill_manual(values = gray(seq(0.3, 1.0, length = 6))) +
  # Removes the axes, the background and legends
  theme_void() +
  theme(legend.position = "none")

We can see that the values for the slope, the intercept, the R2 and the RSE are the same for all four models. That reveals that without visualizing the data, an analyst could make serious modeling mistakes. Plotting the data, however reveals that each data set behaves very differently from the others. Data set 2 and data set 4, for example, are clearly nonlinear, which is prohibitive for regressions.

Another way to see the regressions is using confidence intervals. Once again, a different visualization proves that the data sets are different, regardless of the equivalent metrics:

library(ggplot2)
library(patchwork)

data(anscombe)

plot_anscombe <- function(dataset_num) {
  x_col <- paste0("x", dataset_num)
  y_col <- paste0("y", dataset_num)
  formula <- as.formula(paste(y_col, "~", x_col))
  mod <- lm(formula, data = anscombe)
  intercept <- round(coef(mod)[1], 3)
  slope     <- round(coef(mod)[2], 3)
  r2        <- round(summary(mod)$r.squared, 3)
  rse       <- round(summary(mod)$sigma, 3)
  stats_label <- paste0(
    "Intercept = ", intercept, "\n",
    "Slope = ", slope, "\n",
    "R² = ", r2, "\n",
    "RSE = ", rse
  )
  
  # Building confidence bands
  ggplot(anscombe, aes_string(x = x_col, y = y_col)) +
    geom_smooth(method = "lm", formula = y ~ x, se = TRUE, 
                color = "blue", fill = "lightblue", alpha = 0.4) +
    geom_point(fill = "tomato", color = "red", shape = 21, size = 3) +
    annotate("text", x = 3.5, y = 11.5, label = stats_label, 
             hjust = 0, size = 3.2, family = "sans") +
    scale_x_continuous(limits = c(3, 19)) +
    scale_y_continuous(limits = c(3, 13)) +
    labs(title = paste("Dataset", dataset_num), x = paste0("x", dataset_num), y = paste0("y", dataset_num)) +
    theme_bw()
}

p1 <- plot_anscombe(1)
p2 <- plot_anscombe(2)
p3 <- plot_anscombe(3)
p4 <- plot_anscombe(4)

(p1 | p2) / (p3 | p4) + 
  plot_annotation(
    title = "Anscombe's 4 Regression Data Sets (with 95% CI Bands)",
    theme = theme(plot.title = element_text(size = 16, face = "bold", hjust = 0.5))
  )

4. Now in ggplot2

library(ggplot2)
library(tidyr)
library(dplyr)

# 1. Preparação dos dados em formato longo (tidy format)
data(VADeaths)

df_vadeaths <- as.data.frame(VADeaths) %>%
  mutate(Age = rownames(VADeaths)) %>%
  pivot_longer(cols = -Age, names_to = "Group", values_to = "Deaths") %>%
  mutate(
    Age = factor(Age, levels = rev(rownames(VADeaths))), # Mantém a ordem do empilhamento
    Group = sub(" ", "\n", Group)
  )

# Totais por grupo para posicionar o caracter de plotagem no topo de cada barra
df_totals <- df_vadeaths %>%
  group_by(Group) %>%
  summarise(Total = sum(Deaths))

# Paleta de cores personalizada (não padrão)
custom_cols <- c("#D55E00", "#E69F00", "#68BB68", "#4682B4", "#2B5C8F")

# 2. Plotagem com ggplot2
ggplot(df_vadeaths, aes(x = Group, y = Deaths, fill = Age)) +
  # Barras empilhadas com borda fina
  geom_bar(stat = "identity", position = "stack", color = "white", width = 0.6) +
  # Texto numérico dentro dos segmentos das barras
  geom_text(aes(label = Deaths), 
            position = position_stack(vjust = 0.5), 
            color = "white", fontface = "bold", size = 3.5, family = "serif") +
  # c. Own plotting character (Diamantes/pch=23 no topo das barras)
  geom_point(data = df_totals, aes(x = Group, y = Total + 6, fill = NULL), 
             shape = 23, fill = "#D55E00", color = "black", size = 3.5, stroke = 1) +
  # b. Non-default colours
  scale_fill_manual(values = custom_cols) +
  scale_y_continuous(limits = c(0, 225)) +
  labs(
    title = "Death Rates in Virginia (1940)",
    x = NULL,
    y = "Deaths per 1000",
    fill = "Age Group"
  ) +
  theme_minimal() +
  # a. Use a serif font globally
  theme(
    text = element_text(family = "serif"),
    plot.title = element_text(face = "bold", size = 14, color = "darkblue", hjust = 0.5),
    axis.title.y = element_text(size = 11, color = "gray10"),
    axis.text = element_text(size = 10, color = "gray20"),
    legend.position = "right"
  )
