1 Changing the font in two ways

a. With CSS

library(shiny)
library(datasets)

# Data pre-processing ----
# Nicer labels for the transmission variable (done once, at startup)
mpgData <- mtcars
mpgData$am <- factor(mpgData$am, labels = c("Automatic", "Manual"))


# UI ----
ui <- fluidPage(

  # >>> NEW (1b-i): CSS style block with my own fonts ----
  tags$head(tags$style(HTML("
    body {
      background-color: white;
      color: black;
      font-family: 'Palatino Linotype', Palatino, serif;
    }
    h1, h2, h3 {
      font-family: 'Gill Sans', 'Gill Sans MT', sans-serif;
    }
    .shiny-input-container {
      color: #000000;
    }
  "))),
  # h3 was added to the heading rule so the caption above the plot
  # (h3(textOutput(...))) uses the heading font too.

  # App title (an h2, so it gets the heading font) ----
  titlePanel("Miles Per Gallon"),

  sidebarLayout(

    # Sidebar panel for inputs ----
    sidebarPanel(
      # Input: variable to plot against mpg
      selectInput("variable", "Variable:",
                  c("Cylinders" = "cyl",
                    "Transmission" = "am",
                    "Gears" = "gear")),
      # Input: whether outliers are shown
      checkboxInput("outliers", "Show outliers", TRUE)
    ),

    # Main panel for outputs ----
    mainPanel(
      h3(textOutput("caption")),   # formula used as caption
      plotOutput("mpgPlot")        # box plot
    )
  )
)


# Server ----
server <- function(input, output) {

  # Formula text, shared by the caption and the plot
  formulaText <- reactive({
    paste("mpg ~", input$variable)
  })

  output$caption <- renderText({
    formulaText()
  })

  output$mpgPlot <- renderPlot({
    # CSS only styles the web page; the plot is an image drawn by R.
    # >>> NEW (1b-i): ask R for its generic serif font so the plot matches the body text
    par(family = "serif")
    boxplot(as.formula(formulaText()),
            data = mpgData,
            outline = input$outliers,
            col = "#75AADB", pch = 19,
            ylab = "Miles per gallon")
  })
}


# Run the app ----
shinyApp(ui, server)

b. With bslib

library(shiny)
library(datasets)
library(bslib)
library(thematic)

# Data pre-processing ----
mpgData <- mtcars
mpgData$am <- factor(mpgData$am, labels = c("Automatic", "Manual"))


# UI ----
ui <- fluidPage(

  # >>> NEW (1b-ii): Bootstrap theme with a Google font ----
  theme = bs_theme(
    base_font    = font_google("Lora"),   # body text and inputs
    heading_font = font_google("Lora")    # titles (optional: could be another font)
  ),

  titlePanel("Miles Per Gallon"),

  sidebarLayout(
    sidebarPanel(
      selectInput("variable", "Variable:",
                  c("Cylinders" = "cyl",
                    "Transmission" = "am",
                    "Gears" = "gear")),
      checkboxInput("outliers", "Show outliers", TRUE)
    ),
    mainPanel(
      h3(textOutput("caption")),
      plotOutput("mpgPlot")
    )
  )
)


# Server ----
server <- function(input, output, session) {

  # >>> NEW (1b-ii): plots follow the theme's font and colours.
  # font = "auto" is required: the default (font = NA) leaves the plot font unchanged.
  thematic_shiny(font = "auto")

  formulaText <- reactive({
    paste("mpg ~", input$variable)
  })

  output$caption <- renderText({
    formulaText()
  })

  output$mpgPlot <- renderPlot({
    boxplot(as.formula(formulaText()),
            data = mpgData,
            outline = input$outliers,
            col = "#75AADB", pch = 19,
            ylab = "Miles per gallon")
  })
}


# Run the app ----
shinyApp(ui, server)

c. Which I would use

For a production project, I would use bslib. It allows us to manage fonts and colors centrally for all Bootstrap components. font_google() bundles the font directly with the application so every user sees the same typeface regardless of their system settings, while thematic_shiny(font = “auto”) applies that same font to R plots, which CSS cannot styling directly. I prefer to keep custom CSS strictly for minor, targeted adjustments, such as styling a specific element.

server <- function(input, output, session) {
  # Reactive expression shared across all three outputs
  filtered_cars <- reactive({
    mtcars %>%
      filter(hp >= input$hp_range[1] & hp <= input$hp_range[2],
             gear %in% as.numeric(input$gear_choice))
  })
  
  output$carPlot <- renderPlot({
    df <- filtered_cars()
    ggplot(df, aes(x = wt, y = mpg, color = factor(gear))) +
      geom_point(size = 4) +
      labs(
        title = "Fuel Economy vs. Vehicle Weight",
        subtitle = "Filtered by user horsepower and gear selections",
        x = "Vehicle Weight (1000 lbs)",
        y = "Fuel Economy (Miles Per Gallon)",
        color = "Gears",
        caption = "Source: 1974 Motor Trend US magazine (mtcars dataset)"
      ) +
      theme_minimal()
  })
  
  output$carTable <- renderTable({
    filtered_cars() %>% tibble::rownames_to_column("Model") %>% select(Model, mpg, cyl, disp, hp, wt, gear)
  })
  
  output$carSummary <- renderPrint({
    summary(filtered_cars()[, c("mpg", "hp", "wt", "disp")])
  })
}

server <- function(input, output, session) {
  # Reactive expression shared by plot and table
  processed_arrests <- reactive({
    df <- USArrests %>% tibble::rownames_to_column("State")
    
    if (input$sort_order == "desc") {
      df %>% arrange(desc(.data[[input$crime_type]]))
    } else {
      df %>% arrange(State)
    }

  output$arrestPlot <- renderPlot({
    df <- processed_arrests()
    
    y_label_text <- switch(input$crime_type,

# =============================================================================
# SERVER
# =============================================================================
server <- function(input, output, session) {

  # ---- One reactive with every number the outputs need ----------------------
  res <- reactive({
    stat <- input$stat
    a    <- as.numeric(input$alpha)
    tail <- input$tail
    sv   <- input[[paste0("sv_", stat)]]
    h0   <- input[[paste0("h0_", stat)]]
    se   <- input[[paste0("se_", stat)]]
    req(sv, h0, se)

    kind <- if (stat == "var") "chi" else if (stat == "mean" && input$sigma == "unknown") "t" else "z"
    df   <- switch(kind, chi = se - 1, t = input$n_t - 1, z = NA)

    # density, cumulative probability and quantile of the reference distribution
    d <- switch(kind,
      z   = list(pdf = dnorm, cdf = pnorm, q = qnorm, L = "z", RV = "Z"),
      t   = list(pdf = \(x) dt(x, df), cdf = \(x) pt(x, df), q = \(p) qt(p, df),
                 L = "t", RV = sprintf("T_{%d}", df)),
      chi = list(pdf = \(x) dchisq(x, df), cdf = \(x) pchisq(x, df), q = \(p) qchisq(p, df),
                 L = "\\chi^2", RV = sprintf("\\chi^2_{%d}", df)))

    ts <- if (kind == "chi") df * sv / h0 else (sv - h0) / se      # test statistic

    # critical values (cut-offs of the rejection region)
    lo <- hi <- NA
    if (tail == "two") { lo <- d$q(a / 2); hi <- d$q(1 - a / 2) }
    if (tail == "left")  lo <- d$q(a)
    if (tail == "right") hi <- d$q(1 - a)

    # p-value and cut-offs of the P region
    F <- d$cdf(ts); plo <- phi <- NA
    if (tail == "two") { pt_ <- min(F, 1 - F); p <- min(1, 2 * pt_); plo <- d$q(pt_); phi <- d$q(1 - pt_) }
    if (tail == "left")  { p <- F;     plo <- ts }
    if (tail == "right") { p <- 1 - F; phi <- ts }

    # x-axis range
    if (kind == "chi") {
      xr <- c(0, min(max(d$q(0.9995), ts * 1.08), d$q(0.999999)))
    } else {
      m  <- min(max(4, abs(ts) + 0.6, 1.15 * max(abs(c(lo, hi)), na.rm = TRUE)), 8)
      xr <- c(-m, m)
    }

    list(stat = stat, a = a, tail = tail, sv = sv, h0 = h0, se = se, kind = kind, df = df,
         d = d, ts = ts, lo = lo, hi = hi, F = F, p = p, plo = plo, phi = phi, xr = xr,
         reject = p < a, dg = CONF[[stat]]$d,
         center = if (kind == "chi") df else 0,
         center_txt = if (kind == "chi") paste("df =", df) else "0")
  })

  # ---- Draws one distribution card ------------------------------------------
  # regions: list of c(from, to) shaded with `fill`; ticks: values labelled under the axis;
  # marks: values with a short vertical tick; corner: labels in the top corners
  draw_dist <- function(R, regions = list(), fill = NA, ticks = NULL, tick_col = COL$grey,
                        marks = NULL, mark_col = COL$grey, corner = list()) {
    xs   <- seq(R$xr[1], R$xr[2], length.out = 500)
    dat  <- data.frame(x = xs, y = R$d$pdf(xs))
    ymax <- max(dat$y) * 1.32
    g <- ggplot(dat, aes(x, y))
    for (r in regions) {
      sub <- dat[dat$x >= r[1] & dat$x <= r[2], ]
      if (nrow(sub) > 1) g <- g + geom_area(data = sub, fill = fill)
    }
    g <- g + geom_line(colour = COL$curve, linewidth = 0.7) +
      geom_hline(yintercept = 0, colour = "#c9d0d7")
    for (v in marks[!is.na(marks)]) if (v >= R$xr[1] && v <= R$xr[2])
      g <- g + annotate("segment", x = v, xend = v, y = 0, yend = ymax * 0.12, colour = mark_col, linewidth = 0.8)
    # labels under the axis (an arrow shows a value outside the plotted range)
    lab <- data.frame(v = c(R$center, ticks), txt = c(R$center_txt, fm(ticks, 2)),
                      col = c(COL$grey, rep(tick_col, length(ticks))), face = c("plain", rep("bold", length(ticks))))
    lab <- lab[!is.na(lab$v), ]
    lab$txt[lab$v < R$xr[1]] <- paste("\u2190", lab$txt[lab$v < R$xr[1]])
    lab$txt[lab$v > R$xr[2]] <- paste(lab$txt[lab$v > R$xr[2]], "\u2192")
    lab$v <- pmin(pmax(lab$v, R$xr[1]), R$xr[2])
    g <- g + annotate("text", x = lab$v, y = -ymax * 0.07, label = lab$txt, colour = lab$col,
                      fontface = lab$face, size = 3.6)
    for (cl in corner)
      g <- g + annotate("text", x = if (cl$side == "left") R$xr[1] else R$xr[2], y = ymax * 0.97,
                        label = cl$text, hjust = if (cl$side == "left") 0 else 1,
                        colour = cl$col, fontface = "bold", size = 4)
    g + coord_cartesian(xlim = R$xr, ylim = c(-ymax * 0.12, ymax), expand = FALSE, clip = "off") +
      theme_void()
  }

  # P-region pieces (used by the test statistic and p-value cards)
  p_parts <- reactive({
    R <- res()
    if (R$tail == "two")  list(reg = list(c(-Inf, R$plo), c(R$phi, Inf)), ticks = c(R$plo, R$phi),
                               corner = list(list(side = "left", text = fp(R$p / 2), col = COL$p),
                                             list(side = "right", text = fp(R$p / 2), col = COL$p)))
    else if (R$tail == "left") list(reg = list(c(-Inf, R$plo)), ticks = R$plo,
                               corner = list(list(side = "left", text = fp(R$p), col = COL$p)))
    else list(reg = list(c(R$phi, Inf)), ticks = R$phi,
              corner = list(list(side = "right", text = fp(R$p), col = COL$p)))
  })
  a_regions <- reactive({
    R <- res()
    Filter(Negate(is.null), list(if (!is.na(R$lo)) c(-Inf, R$lo), if (!is.na(R$hi)) c(R$hi, Inf)))
  })

  # ---- Hypotheses and help text ---------------------------------------------
  par_tex <- reactive(switch(input$stat, mean = "\\mu", prop = "p", var = "\\sigma^2"))
  output$hyp <- renderUI({
    R <- res(); op <- switch(R$tail, two = "\\neq", left = "<", right = ">")
    v <- fx(R$h0, R$dg[["h0"]])
    mj("H_0:\\ ", par_tex(), " = ", v, "\\qquad H_1:\\ ", par_tex(), " ", op, " ", v)
  })
  output$se_help <- renderUI({
    R <- res()
    switch(R$kind,
      z   = if (R$stat == "mean") mj("\\text{Standard error under } H_0:\\ SE = \\dfrac{\\sigma}{\\sqrt{n}} = ", fx(R$se, 2))
            else mj("SE = \\sqrt{\\dfrac{p_0(1-p_0)}{n}} = ", fx(R$se, 3),
                    "\\quad (\\text{implies } n \\approx ", round(R$h0 * (1 - R$h0) / R$se^2), ")"),
      t   = mj("SE = \\dfrac{s}{\\sqrt{n}} = ", fx(R$se, 2), ",\\quad df = n - 1 = ", R$df),
      chi = mj("\\text{The test uses } \\chi^2 \\text{ with } df = n - 1 = ", R$df))
  })

  # ---- Test statistic card --------------------------------------------------
  output$tstat_sub <- renderText({
    R <- res(); switch(R$kind, z = "standardized score", t = paste0("t score, df = ", R$df), chi = "chi-square score")
  })
  output$tstat_eq <- renderUI({
    R <- res(); dg <- R$dg
    if (R$kind == "chi")
      mj("\\chi^2 = \\dfrac{(n-1)\\,s^2}{\\sigma_0^2} = \\dfrac{", R$df, " \\times ", fx(R$sv, 0), "}{", fx(R$h0, 0), "} = ", fx(R$ts, 2))
    else {
      s <- if (R$stat == "mean") c("\\bar{x}", "\\mu_0") else c("\\hat{p}", "p_0")
      mj(R$d$L, " = \\dfrac{", s[1], " - ", s[2], "}{SE} = \\dfrac{", fx(R$sv, dg[["sv"]]), " - ",
         fx(R$h0, dg[["h0"]]), "}{", fx(R$se, dg[["se"]]), "} = ", fx(R$ts, 2))
    }
  })
  output$p_tstat <- renderPlot({
    R <- res(); P <- p_parts()
    draw_dist(R, ticks = P$ticks, tick_col = COL$p, marks = P$ticks, mark_col = COL$p)   # curve + marks only
  })

  # ---- P-value card ---------------------------------------------------------
  output$pval_sub <- renderText({
    R <- res()
    if (R$tail == "two") { if (R$kind == "chi") "area in both tails, doubled from the nearer one"
                           else paste0("area beyond \u00B1", fx(abs(R$ts), 2)) }
    else paste("area", if (R$tail == "left") "below" else "above", fx(R$ts, 2))
  })
  output$pval_eq <- renderUI({
    R <- res(); t2 <- fx(R$ts, 2)
    if (R$tail == "two") {
      side <- if (R$kind == "chi") paste(if (R$F < 0.5) "\\le" else "\\ge", t2) else paste("\\ge", fx(abs(R$ts), 2))
      mj("P = 2 \\cdot P(", R$d$RV, " ", side, ") ", fp_eq(R$p))
    } else mj("P = P(", R$d$RV, if (R$tail == "left") " \\le " else " \\ge ", t2, ") ", fp_eq(R$p))
  })
  output$p_pval <- renderPlot({
    R <- res(); P <- p_parts()
    draw_dist(R, regions = P$reg, fill = COL$p_fill, ticks = P$ticks, tick_col = COL$p,   # orange area only
              corner = list(list(side = if (R$tail == "right") "right" else "left",
                                 text = paste("P =", fp(R$p)), col = COL$p)))
  })

  # ---- Critical value card --------------------------------------------------
  output$crit_sub <- renderText({
    R <- res(); w <- switch(R$kind, z = "z", t = "t", chi = "\u03C7\u00B2")
    dft <- if (R$kind == "t") paste0(" (df = ", R$df, ")") else ""
    if (R$tail == "two") { if (R$kind == "chi") "values that cut off \u03B1/2 in each tail" else paste0(w, " that cuts off those tails", dft) }
    else paste0(w, " that cuts off the ", R$tail, " tail", dft)
  })
  output$crit_eq <- renderUI({
    R <- res()
    if (R$tail == "two") {
      if (R$kind == "chi") mj("\\chi^2_L = ", fx(R$lo, 2), ",\\quad \\chi^2_U = ", fx(R$hi, 2))
      else mj(R$d$L, " = \\pm ", fx(R$hi, 2))
    } else mj(R$d$L, " = ", fx(if (R$tail == "left") R$lo else R$hi, 2))
  })
  output$p_crit <- renderPlot({
    R <- res(); cv <- c(R$lo, R$hi); cv <- cv[!is.na(cv)]
    draw_dist(R, ticks = cv, tick_col = COL$a, marks = cv, mark_col = COL$a)   # curve + marks only
  })

  # ---- Alpha tail area card -------------------------------------------------
  output$atail_sub <- renderText({
    R <- res(); if (R$tail == "two") "area split over two tails" else paste("all of \u03B1 in the", R$tail, "tail")
  })
  output$atail_eq <- renderUI({
    R <- res()
    if (R$tail == "two") mj("\\alpha = ", R$a, ",\\qquad \\dfrac{\\alpha}{2} = ", signif(R$a / 2, 4), "\\text{ per tail}")
    else mj("\\alpha = ", R$a, "\\text{ in the ", R$tail, " tail}")
  })
  output$p_atail <- renderPlot({
    R <- res(); cv <- c(R$lo, R$hi); cv <- cv[!is.na(cv)]
    draw_dist(R, regions = a_regions(), fill = COL$a_fill, ticks = cv, tick_col = COL$a)   # blue area only
  })

  # ---- Decision -------------------------------------------------------------
  output$decision <- renderUI({
    R <- res(); L <- R$d$L; t2 <- fx(R$ts, 2)
    par_txt <- switch(R$stat, mean = "\u03BC", prop = "p", var = "\u03C3\u00B2")
    op_txt  <- switch(R$tail, two = "\u2260", left = "<", right = ">")
    v <- fx(R$h0, R$dg[["h0"]])

    rule_p <- if (R$p < 1e-4) paste0("P < 0.0001 < \\alpha = ", R$a)
              else paste0("P = ", fp(R$p), if (R$reject) " < " else " \\ge ", "\\alpha = ", R$a)
    rule_z <- if (R$tail == "two" && R$kind != "chi") {
      paste0("|", L, "| = ", fx(abs(R$ts), 2), if (R$reject) " > " else " \\le ", fx(R$hi, 2))
    } else if (R$tail == "two") {
      if (!R$reject) paste0(fx(R$lo, 2), " \\le \\chi^2 = ", t2, " \\le ", fx(R$hi, 2))
      else if (R$ts > R$hi) paste0("\\chi^2 = ", t2, " > ", fx(R$hi, 2))
      else paste0("\\chi^2 = ", t2, " < ", fx(R$lo, 2))
    } else if (R$tail == "left") paste0(L, " = ", t2, if (R$reject) " < " else " \\ge ", fx(R$lo, 2))
    else paste0(L, " = ", t2, if (R$reject) " > " else " \\le ", fx(R$hi, 2))

    layout_columns(
      col_widths = c(4, 4, 4),
      div(class = paste("verdict", if (R$reject) "rej" else "acc"),
          div(class = "big", style = paste0("color:", if (R$reject) "#a63603" else COL$a),
              HTML(if (R$reject) "Reject H<sub>0</sub>" else "Fail to reject H<sub>0</sub>")),
          div(style = "font-size: 13px; color: #5f6b7a;",
              HTML(if (R$reject) sprintf("At \u03B1 = %s, the sample gives significant evidence that %s %s %s.", R$a, par_txt, op_txt, v)
                   else sprintf("At \u03B1 = %s, the sample does not give enough evidence that %s %s %s. This does not prove H<sub>0</sub> is true.", R$a, par_txt, op_txt, v)))),
      div(class = "rule", div(class = "t", "Compare areas"), mj(rule_p),
          div(class = "r", if (R$reject) "the data's tail is smaller than \u03B1" else "the data's tail is at least as big as \u03B1")),
      div(class = "rule", div(class = "t", "Compare scores"), mj(rule_z),
          div(class = "r", if (R$reject) "the test statistic falls in the rejection region" else "the test statistic is outside the rejection region"))
    )
  })
}

shinyApp(ui, server)
